# 深色模式 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为 Popup 增加可持久化的手动深色模式，覆盖预览、目录、编辑面板、搜索条、菜单栏、托盘菜单、状态栏、滚动条、对话框与系统标题栏。

**Architecture:** 把现有内联 QSS 与 `wrap_html` 的 CSS 颜色抽成模块级色表（`LIGHT`/`DARK`）与接收颜色字典的样式函数。应用级深浅色交给 Qt 6.8+ 的 `QApplication.styleHints().setColorScheme`，不切换风格、不手写调色板、不写 DWM。切换主题时重设各控件样式表并重渲染预览（保留滚动位置与锚点映射）。

**Tech Stack:** Python 3.14、PySide6（`QApplication.styleHints().setColorScheme` / `QToolButton` / `QTextBrowser`）、python-markdown、pytest（unittest 写法）。

**Spec:** `docs/superpowers/specs/2026-10-26-dark-mode-design.md`

## Global Constraints

- 代码只改 `md_viewer.py`；测试新增 `tests/test_theme.py`；依赖改 `requirements.txt`；文档改 `README.md`、`helpdocs/使用手册.md`。
- `setColorScheme` 自 PySide6 6.8 起提供，`requirements.txt` 由 `PySide6>=6.5` 提升为 `PySide6>=6.8`。
- 应用级主题**不切换风格**（保持 `windows11`/offscreen 的 `fusion`）、**不手写 `QPalette`**、**不写 ctypes/DWM**。
- `QSettings` 键 `appearance/dark_mode`，类型 bool，默认 `False`。
- 测试文件首行区设置 `QT_QPA_PLATFORM=offscreen`；offscreen 不支持配色方案请求，测试断言 mock 的 `setColorScheme` 调用参数，不断言真实调色板或像素。
- 运行全部测试：`python -m pytest tests -q`（基线 92 passed）。
- 提交信息用中文，前缀 `feat:` / `test:` / `refactor:` / `docs:`。

import os
import unittest
from unittest import mock

os.environ.setdefault("QT_QPA_PLATFORM", "offscreen")

from PySide6.QtGui import QIcon
from PySide6.QtWidgets import QApplication

from md_viewer import (
    DARK,
    LIGHT,
    MarkdownViewer,
    _build_pin_icon,
    _search_input_qss,
)

_app = QApplication.instance() or QApplication([])


class TestThemePalette(unittest.TestCase):
    def test_light_dark_same_keys(self):
        self.assertEqual(set(LIGHT.keys()), set(DARK.keys()))

    def test_wrap_html_light_contains_existing_colors(self):
        html = MarkdownViewer.wrap_html("<p>x</p>", LIGHT)
        for needle in (
            "background: #fff",
            "color: #333",
            "color: #0366d6",
            "background: #f4f4f4",
            "background-color: #e8e8e8",
        ):
            self.assertIn(needle, html)

    def test_wrap_html_dark_uses_dark_colors(self):
        html = MarkdownViewer.wrap_html("<p>x</p>", DARK)
        self.assertIn(DARK["bg"], html)
        self.assertIn(DARK["fg"], html)
        self.assertNotIn("background: #fff", html)


class TestThemeQss(unittest.TestCase):
    def test_search_input_qss_error(self):
        qss = _search_input_qss(LIGHT, error=True)
        self.assertIn(LIGHT["error_border"], qss)

    def test_search_input_qss_normal(self):
        qss = _search_input_qss(LIGHT, error=False)
        self.assertIn(LIGHT["input_border"], qss)


class TestPinIconTheme(unittest.TestCase):
    def test_pin_icon_on_bright_in_dark(self):
        icon = _build_pin_icon(DARK["pin_on"])
        pixmap = icon.pixmap(16, 16, QIcon.Mode.Normal, QIcon.State.On)
        self.assertFalse(pixmap.isNull())
        image = pixmap.toImage()
        total = 0
        count = 0
        for x in range(image.width()):
            for y in range(image.height()):
                c = image.pixelColor(x, y)
                if c.alpha() > 0:
                    total += (c.red() + c.green() + c.blue()) / 3
                    count += 1
        self.assertGreater(count, 0)
        self.assertGreater(total / count, 128)


if __name__ == "__main__":
    unittest.main()

- [ ] **Step 4: 新增 `apply_theme` 与 `toggle_dark_mode`**

```python
    def apply_theme(self, dark):
        """按深/浅色重设应用配色方案、各控件样式并重渲染预览。"""
        self._colors = DARK if dark else LIGHT
        QApplication.instance().styleHints().setColorScheme(
            Qt.ColorScheme.Dark if dark else Qt.ColorScheme.Light
        )
        self.toc_tree.setStyleSheet(_toc_qss(self._colors))
        self.text_browser.setStyleSheet(_browser_qss(self._colors))
        self.toc_toggle_btn.setStyleSheet(_toc_toggle_qss(self._colors))
        self.search_bar.setStyleSheet(_search_bar_qss(self._colors))
        self.search_input.setStyleSheet(
            _search_input_qss(self._colors, error=self.match_count_label.text() == "无结果")
        )
        self.match_count_label.setStyleSheet(_muted_label_qss(self._colors))
        self.case_btn.setStyleSheet(_option_btn_qss(self._colors))
        self.word_btn.setStyleSheet(_option_btn_qss(self._colors))
        self.pin_btn.setStyleSheet(_pin_btn_qss(self._colors))
        if self.file_path:
            self.reload_file()
        else:
            self._show_welcome()
        self.pin_btn.setIcon(_build_pin_icon(self._colors["pin_on"]))

    def toggle_dark_mode(self, checked):
        """切换深色模式并持久化。"""
        self._dark_mode = checked
        QSettings().setValue("appearance/dark_mode", checked)
        self.apply_theme(checked)
```

```python
class TestApplyTheme(unittest.TestCase):
    def setUp(self):
        # 隔离构造副作用：不装全局键盘钩子、不把 pytest 参数当文件打开、不读写用户 QSettings
        patcher_hotkey = mock.patch("md_viewer.GlobalHotkey")
        patcher_hotkey.start()
        self.addCleanup(patcher_hotkey.stop)
        patcher_argv = mock.patch.object(sys, "argv", ["md_viewer.py"])
        patcher_argv.start()
        self.addCleanup(patcher_argv.stop)
        patcher_settings = mock.patch(
            "md_viewer.QSettings",
            return_value=mock.MagicMock(value=mock.MagicMock(return_value=False)),
        )
        patcher_settings.start()
        self.addCleanup(patcher_settings.stop)

        self.viewer = MarkdownViewer()
        self.addCleanup(self.viewer.close)
        self.style_hints = mock.MagicMock()
        patcher_hints = mock.patch.object(
            QApplication.instance(), "styleHints", return_value=self.style_hints
        )
        patcher_hints.start()
        self.addCleanup(patcher_hints.stop)
        self.orig_style = QApplication.instance().style().name()

    def test_apply_theme_toggles_dark_and_light(self):
        self.viewer.apply_theme(True)
        self.style_hints.setColorScheme.assert_called_with(Qt.ColorScheme.Dark)
        self.assertIn(DARK["toc_bg"], self.viewer.toc_tree.styleSheet())
        self.assertEqual(QApplication.instance().style().name(), self.orig_style)

        self.viewer.apply_theme(False)
        self.style_hints.setColorScheme.assert_called_with(Qt.ColorScheme.Light)
        self.assertIn(LIGHT["toc_bg"], self.viewer.toc_tree.styleSheet())
```
